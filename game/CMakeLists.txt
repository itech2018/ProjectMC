include(FetchContent)

# Use SDL's release archive rather than a shallow Git checkout.
# This avoids FetchContent's Git update step, which can break on some
# Windows/CMake/Git combinations with errors such as an invalid HEAD0 ref.
FetchContent_Declare(SDL3
  URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.16/SDL3-3.4.16.tar.gz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(SDL3)

add_executable(projectmc-game
  src/main.cpp
  src/Application.cpp
  src/ChunkRenderer.cpp
  src/Player.cpp
  src/Raycast.cpp
  src/RenderBackend.cpp
  src/GpuRenderBackend.cpp
  src/GpuShaderLoader.cpp
  src/RenderMath.cpp
  src/SdlRenderBackend.cpp
  src/TextureAtlas.cpp
)
target_include_directories(projectmc-game PRIVATE include)
target_link_libraries(projectmc-game PRIVATE projectmc-shared SDL3::SDL3-static)
target_compile_features(projectmc-game PRIVATE cxx_std_20)


# GPU shader sources are kept as explicit project assets while the renderer is
# migrated. Compiled binaries are generated by the optional shader toolchain,
# not required for the SDL compatibility renderer.
set(PROJECTMC_SHADER_SOURCES
  ${CMAKE_CURRENT_SOURCE_DIR}/shaders/voxel.vert.hlsl
  ${CMAKE_CURRENT_SOURCE_DIR}/shaders/voxel.frag.hlsl
)
target_sources(projectmc-game PRIVATE ${PROJECTMC_SHADER_SOURCES})
source_group("Shaders" FILES ${PROJECTMC_SHADER_SOURCES})

# These are canonical shader sources, not Visual Studio FXC build inputs.
# SDL_shadercross owns compilation so it can emit the backend formats SDL_GPU
# expects. HEADER_FILE_ONLY keeps the sources visible in the IDE without MSVC
# attempting an older default HLSL profile during projectmc-game builds.
set_source_files_properties(${PROJECTMC_SHADER_SOURCES} PROPERTIES
  HEADER_FILE_ONLY TRUE
)

# Compile the Windows SDL_GPU shaders to DXIL with Microsoft's DirectX
# Shader Compiler when it is available. The runtime loader expects these exact
# names under <exe>/shaders.
if(WIN32)
  # Visual Studio developer shells do not always put DXC on PATH. Search the
  # versioned Windows SDK bin directories and prefer the newest x64 compiler.
  file(GLOB PROJECTMC_WINDOWS_SDK_DXC
    "C:/Program Files (x86)/Windows Kits/10/bin/*/x64/dxc.exe"
  )
  if(PROJECTMC_WINDOWS_SDK_DXC)
    list(SORT PROJECTMC_WINDOWS_SDK_DXC COMPARE NATURAL ORDER DESCENDING)
    list(GET PROJECTMC_WINDOWS_SDK_DXC 0 PROJECTMC_DXC)
  else()
    find_program(PROJECTMC_DXC NAMES dxc dxc.exe
      HINTS
        "$ENV{WindowsSdkDir}/bin/$ENV{WindowsSDKVersion}/x64"
        "$ENV{WindowsSdkDir}/bin/x64"
    )
  endif()
else()
  find_program(PROJECTMC_DXC NAMES dxc)
endif()

if(PROJECTMC_DXC)
  message(STATUS "ProjectMC: DXC found: ${PROJECTMC_DXC}")
endif()

set(PROJECTMC_SHADER_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${PROJECTMC_SHADER_OUTPUT_DIR}")

if(PROJECTMC_DXC)
  add_custom_command(
    OUTPUT
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.vert.dxil"
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.frag.dxil"
    COMMAND "${PROJECTMC_DXC}" -T vs_6_0 -E main
      -Fo "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.vert.dxil"
      "${CMAKE_CURRENT_SOURCE_DIR}/shaders/voxel.vert.hlsl"
    COMMAND "${PROJECTMC_DXC}" -T ps_6_0 -E main
      -Fo "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.frag.dxil"
      "${CMAKE_CURRENT_SOURCE_DIR}/shaders/voxel.frag.hlsl"
    DEPENDS ${PROJECTMC_SHADER_SOURCES}
    COMMENT "Compiling ProjectMC voxel shaders to DXIL"
    VERBATIM
  )
  add_custom_target(projectmc-shaders ALL
    DEPENDS
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.vert.dxil"
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.frag.dxil"
  )
  add_dependencies(projectmc-game projectmc-shaders)

  add_custom_command(TARGET projectmc-game POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:projectmc-game>/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.vert.dxil"
      "$<TARGET_FILE_DIR:projectmc-game>/shaders/voxel.vert.dxil"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${PROJECTMC_SHADER_OUTPUT_DIR}/voxel.frag.dxil"
      "$<TARGET_FILE_DIR:projectmc-game>/shaders/voxel.frag.dxil"
    COMMENT "Packaging ProjectMC DXIL shaders"
  )
else()
  add_custom_target(projectmc-shaders
    COMMAND ${CMAKE_COMMAND} -E echo "DXC was not found. Install the DirectX Shader Compiler or place dxc.exe on PATH to build the experimental SDL_GPU renderer."
    SOURCES ${PROJECTMC_SHADER_SOURCES}
    VERBATIM
  )
  message(STATUS "ProjectMC: DXC not found; compatibility renderer will still build.")
endif()
